cmake_minimum_required(VERSION 3.16)
set(OKABE_BOARD "S3" CACHE STRING "Okabe board: S3 or PROTO")
if(OKABE_BOARD STREQUAL "S3")
    set(IDF_TARGET esp32s3)
elseif(OKABE_BOARD STREQUAL "PROTO")
    set(IDF_TARGET esp32)
else()
    message(FATAL_ERROR "OKABE_BOARD must be S3 or PROTO")
endif()
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.defaults.${OKABE_BOARD}")
option(OKABE_RECOVERY "Build the standalone recovery application" OFF)
if("$ENV{OKABE_RECOVERY_PASS}" STREQUAL "1")
    set(OKABE_RECOVERY ON)
endif()
if(OKABE_RECOVERY)
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/recovery")
    # Restrict the build to recovery and its transitive IDF/component dependencies.
    set(COMPONENTS recovery)
    if(DEFINED ENV{PLATFORMIO_BUILD_DIR})
        list(APPEND COMPONENTS __pio_env)
    endif()
else()
    set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/src")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(Okabe_Phone)
